cmake_minimum_required(VERSION 3.20)
project(avic_mqtt_paho_c_worker VERSION 0.1.0 LANGUAGES C)

include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(PAHO_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PAHO_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PAHO_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(PAHO_BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(PAHO_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(PAHO_ENABLE_CPACK OFF CACHE BOOL "" FORCE)
set(PAHO_WITH_SSL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(paho_mqtt_c
  GIT_REPOSITORY https://github.com/eclipse-paho/paho.mqtt.c.git
  GIT_TAG v1.3.16 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(paho_mqtt_c)

find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(JANSSON QUIET IMPORTED_TARGET jansson)
endif()
if(NOT TARGET PkgConfig::JANSSON)
  find_path(JANSSON_INCLUDE_DIR jansson.h REQUIRED)
  find_library(JANSSON_LIBRARY NAMES jansson REQUIRED)
  add_library(jansson_external UNKNOWN IMPORTED)
  set_target_properties(jansson_external PROPERTIES IMPORTED_LOCATION "\${JANSSON_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "\${JANSSON_INCLUDE_DIR}")
  set(JANSSON_TARGET jansson_external)
else()
  set(JANSSON_TARGET PkgConfig::JANSSON)
endif()

add_executable(avic-mqtt-paho-c-worker src/main.c)
target_link_libraries(avic-mqtt-paho-c-worker PRIVATE paho-mqtt3a-static \${JANSSON_TARGET} Threads::Threads)
if(WIN32)
  target_link_libraries(avic-mqtt-paho-c-worker PRIVATE Rpcrt4)
  if(MINGW)
    target_link_options(avic-mqtt-paho-c-worker PRIVATE -static -static-libgcc)
  endif()
else()
  find_library(UUID_LIBRARY NAMES uuid REQUIRED)
  target_link_libraries(avic-mqtt-paho-c-worker PRIVATE \${UUID_LIBRARY})
endif()
if(MSVC)
  target_compile_options(avic-mqtt-paho-c-worker PRIVATE /W4 /WX)
else()
  target_compile_options(avic-mqtt-paho-c-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS avic-mqtt-paho-c-worker RUNTIME DESTINATION bin)
